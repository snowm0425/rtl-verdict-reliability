module multi_8bit (
    input  [7:0] A,
    input  [7:0] B,
    output [15:0] product
);

    // Shift-and-add multiplication
    // For each bit i of B (from LSB to MSB), if B[i] is 1,
    // add A shifted left by i positions to the running product.

    // Combinational implementation using shift-and-add
    // product = sum_{i=0}^{7} (B[i] * A << i)

    // We can implement this as a series of conditional additions
    // or use a simple combinational approach.

    // Method: Build the product by adding shifted versions of A
    // where the shift amount equals the bit position in B.

    // Let's use a straightforward combinational logic approach:
    // product = A * B, but implemented via shift-and-add conceptually.

    // Since this is a combinational circuit, we can express it as:
    // product[15:0] = 
    //   (A[7:0] << 0) * B[0]
    // + (A[7:0] << 1) * B[1]
    // + (A[7:0] << 2) * B[2]
    // + ...
    // + (A[7:0] << 7) * B[7]

    // Each term is an 8-bit value shifted left by i, so it occupies bits [i+7:i]
    // We need to sum these 8 terms, each up to 16 bits wide.

    // Let's define each term:
    wire [15:0] term0;
    wire [15:0] term1;
    wire [15:0] term2;
    wire [15:0] term3;
    wire [15:0] term4;
    wire [15:0] term5;
    wire [15:0] term6;
    wire [15:0] term7;

    // term i = B[i] ? (A << i) : 0

    assign term0 = B[0] ? {A[7:0], 8'd0} : 16'd0;
    assign term1 = B[1] ? {A[6:0], 1'b0, 8'd0} : 16'd0;
    assign term2 = B[2] ? {A[5:0], 2'b00, 8'd0} : 16'd0;
    assign term3 = B[3] ? {A[4:0], 3'b000, 8'd0} : 16'd0;
    assign term4 = B[4] ? {A[3:0], 4'b0000, 8'd0} : 16'd0;
    assign term5 = B[5] ? {A[2:0], 5'b00000, 8'd0} : 16'd0;
    assign term6 = B[6] ? {A[1:0], 6'b000000, 8'd0} : 16'd0;
    assign term7 = B[7] ? {A[0], 7'b0000000, 8'd0} : 16'd0;

    // Now sum all terms
    // We can add them in stages to keep the logic manageable

    wire [15:0] sum01 = term0 + term1;
    wire [15:0] sum23 = term2 + term3;
    wire [15:0] sum45 = term4 + term5;
    wire [15:0] sum67 = term6 + term7;

    wire [15:0] sum0123 = sum01 + sum23;
    wire [15:0] sum4567 = sum45 + sum67;

    assign product = sum0123 + sum4567;

endmodule
