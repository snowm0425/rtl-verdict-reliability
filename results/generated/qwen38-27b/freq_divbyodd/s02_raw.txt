module freq_divbyodd #(
    parameter NUM_DIV = 5
) (
    input  wire clk,
    input  wire rst_n,
    output reg  clk_div
);

    // Check if NUM_DIV is odd
    // For odd division, we need to handle the asymmetric duty cycle
    // The idea: use two counters, one for rising edge, one for falling edge
    // Each counter counts up to (NUM_DIV-1)/2 or similar

    localparam integer HALF_DIV = (NUM_DIV - 1) / 2;
    localparam integer COUNTER_WIDTH = $clog2(NUM_DIV);

    reg [COUNTER_WIDTH-1:0] cnt_rise;
    reg [COUNTER_WIDTH-1:0] cnt_fall;

    reg clk_div1; // Toggles on rising edge of clk
    reg clk_div2; // Toggles on falling edge of clk

    // Counter for rising edge
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n)
            cnt_rise <= {COUNTER_WIDTH{1'b0}};
        else
            cnt_rise <= (cnt_rise == HALF_DIV) ? {COUNTER_WIDTH{1'b0}} : cnt_rise + 1;
    end

    // Counter for falling edge
    always @(negedge clk or negedge rst_n) begin
        if (!rst_n)
            cnt_fall <= {COUNTER_WIDTH{1'b0}};
        else
            cnt_fall <= (cnt_fall == HALF_DIV) ? {COUNTER_WIDTH{1'b0}} : cnt_fall + 1;
    end

    // Toggle clk_div1 on rising edge when counter reaches HALF_DIV
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n)
            clk_div1 <= 1'b0;
        else if (cnt_rise == HALF_DIV)
            clk_div1 <= ~clk_div1;
    end

    // Toggle clk_div2 on falling edge when counter reaches HALF_DIV
    always @(negedge clk or negedge rst_n) begin
        if (!rst_n)
            clk_div2 <= 1'b0;
        else if (cnt_fall == HALF_DIV)
            clk_div2 <= ~clk_div2;
    end

    // Final output is OR of the two divided clocks
    assign clk_div = clk_div1 | clk_div2;

endmodule
