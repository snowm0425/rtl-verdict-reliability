module freq_divbyodd #(
    parameter NUM_DIV = 5
)(
    input  wire clk,
    input  wire rst_n,
    output reg  clk_div
);

    // Calculate the bit width needed for the counter
    localparam COUNTER_WIDTH = $clog2(NUM_DIV);
    
    // Counters for tracking rising and falling edges
    reg [COUNTER_WIDTH-1:0] cnt1;
    reg [COUNTER_WIDTH-1:0] cnt2;
    
    // Two separate clock divider outputs
    reg clk_div1;
    reg clk_div2;
    
    // Half of NUM_DIV (for odd N, floor(N/2))
    localparam HALF_DIV = NUM_DIV / 2;
    
    // Rising edge of clk
    wire clk_rise = clk && !rst_n;
    // Falling edge of clk
    wire clk_fall = !clk && !rst_n;
    
    // Reset condition
    wire reset = !rst_n;
    
    // Counter 1: increments on rising edge
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n)
            cnt1 <= {COUNTER_WIDTH{1'b0}};
        else if (clk_rise)
            cnt1 <= (cnt1 == NUM_DIV - 1) ? {COUNTER_WIDTH{1'b0}} : cnt1 + 1'b1;
    end
    
    // Counter 2: increments on falling edge
    always @(negedge clk or negedge rst_n) begin
        if (!rst_n)
            cnt2 <= {COUNTER_WIDTH{1'b0}};
        else if (clk_fall)
            cnt2 <= (cnt2 == NUM_DIV - 1) ? {COUNTER_WIDTH{1'b0}} : cnt2 + 1'b1;
    end
    
    // Clock divider 1: toggles when cnt1 reaches HALF_DIV
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n)
            clk_div1 <= 1'b0;
        else if (cnt1 == HALF_DIV)
            clk_div1 <= ~clk_div1;
    end
    
    // Clock divider 2: toggles when cnt2 reaches HALF_DIV
    always @(negedge clk or negedge rst_n) begin
        if (!rst_n)
            clk_div2 <= 1'b0;
        else if (cnt2 == HALF_DIV)
            clk_div2 <= ~clk_div2;
    end
    
    // Final output: OR of the two divided clocks
    always @(*) begin
        clk_div = clk_div1 | clk_div2;
    end

endmodule
