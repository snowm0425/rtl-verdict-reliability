module pe (
    input wire clk,
    input wire rst,
    input wire signed [31:0] a,
    input wire signed [31:0] b,
    output reg signed [31:0] c
);

    // The product of two 32-bit signed numbers is 64 bits.
    // We will compute the full 64-bit product, then add to the 32-bit accumulator.
    // Since the accumulator is 32-bit, we will take the lower 32 bits of the sum.
    // This is equivalent to: c = (c + a*b) mod 2^32
    
    // Alternatively, we can just let Verilog handle the truncation naturally.
    // If we write: c <= c + a * b;
    // Verilog will compute a*b as 64 bits (since both are 32-bit signed, the result 
    // is 64-bit signed), then add to c (which is 32-bit, so it gets extended to 64-bit 
    // signed), and then the result is truncated to 32 bits for assignment to c.
    
    // This is the standard behavior and is likely what is expected.
