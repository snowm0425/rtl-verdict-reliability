```verilog
module multi_booth_8bit (
    input  wire       clk,
    input  wire       reset,
    input  wire [7:0] a,
    input  wire [7:0] b,
    output reg  [15:0] p,
    output reg  rdy
);

    // 16-bit registers for multiplier and multiplicand
    reg [15:0] multiplier;
    reg [15:0] multiplicand;
    reg [4:0]  ctr;

    // Reset logic
    always @(posedge clk or posedge reset) begin
        if (reset) begin
            multiplier <= {{8{a[7]}}, a};
            multiplicand <= {{8{b[7]}}, b};
            p <= 16'b0;
            ctr <= 5'b0;
            rdy <= 1'b0;
        end else begin
            if (ctr < 5'd16) begin
                // Left shift multiplicand by 1
                multiplicand <= multiplicand << 1;
                // If current bit of multiplier is 1, add multiplicand to product
                if (multiplier[ctr]) begin
                    p <= p + multiplicand;
                end
                // Increment counter
                ctr <= ctr + 1'b1;
            end else begin
                rdy <= 1'b1;
            end
        end
    end

endmodule
```
