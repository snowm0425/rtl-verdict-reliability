```verilog
module accu (
    input        clk,
    input        rst_n,
    input  [7:0] data_in,
    input        valid_in,
    output       valid_out,
    output [9:0] data_out
);

    // Counter to track how many valid data have been received (0 to 3)
    reg [1:0] counter;
    // Accumulator to hold the running sum
    reg [9:0] accumulator;
    // Output registers
    reg valid_out_reg;
    reg [9:0] data_out_reg;

    // Assign output ports
    assign valid_out = valid_out_reg;
    assign data_out = data_out_reg;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            counter <= 2'd0;
            accumulator <= 10'd0;
            valid_out_reg <= 1'b0;
            data_out_reg <= 10'd0;
        end else begin
            // Default: no output valid
            valid_out_reg <= 1'b0;

            if (valid_in) begin
                if (counter == 2'd3) begin
                    // This is the 4th valid data
                    accumulator <= accumulator + data_in;
                    // Output the accumulated sum
                    data_out_reg <= accumulator + data_in;
                    valid_out_reg <= 1'b1;
                    // Reset counter and accumulator for next set of 4
                    counter <= 2'd0;
                    accumulator <= 10'd0;
                end else begin
                    // Not the 4th data, accumulate and increment counter
                    accumulator <= accumulator + data_in;
                    counter <= counter + 1'b1;
                end
            end
        end
    end

endmodule
```
