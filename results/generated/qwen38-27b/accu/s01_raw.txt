```verilog
module accu (
    input        clk,
    input        rst_n,
    input  [7:0] data_in,
    input        valid_in,
    output reg   valid_out,
    output reg [9:0] data_out
);

    // Accumulator register to hold the running sum
    reg [9:0] accu_sum;
    // Counter to track how many valid inputs have been received (0 to 3)
    reg [1:0] count;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            accu_sum <= 10'd0;
            count    <= 2'd0;
            valid_out <= 1'b0;
            data_out  <= 10'd0;
        end else begin
            // Default: no output valid in this cycle
            valid_out <= 1'b0;

            if (valid_in) begin
                if (count == 2'd3) begin
                    // Fourth valid input: output the accumulated sum
                    data_out  <= accu_sum + {2'b00, data_in};
                    valid_out <= 1'b1;
                    // Reset accumulator and counter for next batch
                    accu_sum  <= 10'd0;
                    count     <= 2'd0;
                end else begin
                    // Accumulate the current input
                    accu_sum <= accu_sum + {2'b00, data_in};
                    count    <= count + 2'd1;
                end
            end
            // If valid_in is 0, do nothing (hold current state)
        end
    end

endmodule
```
