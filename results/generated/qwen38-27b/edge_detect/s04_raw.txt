```verilog
module edge_detect (
    input  wire clk,
    input  wire rst_n,
    input  wire a,
    output wire rise,
    output wire down
);

    reg a_q;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n)
            a_q <= 1'b0;
        else
            a_q <= a;
    end

    assign rise = (a == 1'b1) && (a_q == 1'b0);
    assign down = (a == 1'b0) && (a_q == 1'b1);

endmodule
```
