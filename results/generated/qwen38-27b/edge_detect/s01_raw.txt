```verilog
module edge_detect (
    input  wire clk,
    input  wire rst_n,
    input  wire a,
    output reg  rise,
    output reg  down
);

    reg a_reg;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            a_reg <= 1'b0;
            rise  <= 1'b0;
            down  <= 1'b0;
        end else begin
            // Update the registered version of 'a'
            a_reg <= a;
            
            // Detect rising edge: a goes from 0 to 1
            if (a_reg == 1'b0 && a == 1'b1)
                rise <= 1'b1;
            else
                rise <= 1'b0;
            
            // Detect falling edge: a goes from 1 to 0
            if (a_reg == 1'b1 && a == 1'b0)
                down <= 1'b1;
            else
                down <= 1'b0;
        end
    end

endmodule
```
