    reg [15:0] breg;   // Register to hold multiplier

    assign yout = yout_r;
    assign done = done_r;

            breg <= 16'b0;

            if (start) begin

                    i <= i; // Hold at 17 until start is deasserted
                end
            end else begin
                i <= 0;
            end

            if (i == 5'd16) begin
                done_r <= 1'b1;

                done_r <= 1'b0;
            end else begin

                        yout_r <= yout_r + ({breg, 16'b0} << (i-1));
