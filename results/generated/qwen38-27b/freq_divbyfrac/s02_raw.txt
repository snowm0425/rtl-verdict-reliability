module freq_divbyfrac (
    input  wire clk,
    input  wire rst_n,
    output reg  clk_div
);

    // 3.5x division: use 7-cycle counter for double-edge technique
    // We need to generate a clock that toggles every 3.5 clock cycles on average.
    // Using double-edge (both rising and falling) clocking, we can achieve half-integer division.

    // Counter to track 7 cycles (since 3.5 * 2 = 7)
    reg [2:0] counter;

    // Intermediate signals for generating the fractional clock
    reg clk_div_a;
    reg clk_div_b;

    // Reset logic
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            counter <= 3'd0;
            clk_div_a <= 1'b0;
            clk_div_b <= 1'b0;
            clk_div <= 1'b0;
        end else begin
            // Increment counter modulo 7
            if (counter == 3'd6)
                counter <= 3'd0;
            else
                counter <= counter + 3'd1;

            // Generate clk_div_a: high for 4 cycles, low for 3 cycles
            // Pattern over 7 cycles: 1111000 (high for first 4, low for next 3)
            case (counter)
                3'd0: clk_div_a <= 1'b1;
                3'd1: clk_div_a <= 1'b1;
                3'd2: clk_div_a <= 1'b1;
                3'd3: clk_div_a <= 1'b1;
                3'd4: clk_div_a <= 1'b0;
                3'd5: clk_div_a <= 1'b0;
                3'd6: clk_div_a <= 1'b0;
                default: clk_div_a <= 1'b0;
            endcase

            // Generate clk_div_b: phase-shifted version, high for 3 cycles, low for 4 cycles
            // Pattern over 7 cycles: 1110001 (shifted by 1 cycle from clk_div_a)
            case (counter)
                3'd0: clk_div_b <= 1'b1;
                3'd1: clk_div_b <= 1'b1;
                3'd2: clk_div_b <= 1'b1;
                3'd3: clk_div_b <= 1'b0;
                3'd4: clk_div_b <= 1'b0;
                3'd5: clk_div_b <= 1'b0;
                3'd6: clk_div_b <= 1'b1;
                default: clk_div_b <= 1'b0;
            endcase

            // OR the two intermediate clocks to produce the fractional divided clock
            clk_div <= clk_div_a | clk_div_b;
        end
    end

endmodule
