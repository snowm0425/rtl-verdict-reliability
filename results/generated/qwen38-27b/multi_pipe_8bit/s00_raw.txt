module multi_pipe_8bit (
    input        clk,
    input        rst_n,
    input        mul_en_in,
    input  [7:0] mul_a,
    input  [7:0] mul_b,
    output       mul_en_out,
    output [15:0] mul_out
);

    // Registers for input enable pipeline
    reg mul_en_out_reg;

    // Registers for input data pipeline
    reg [7:0] mul_a_reg;
    reg [7:0] mul_b_reg;

    // Partial product wires
    wire [15:0] temp [0:7];

    // Partial sum registers
    reg [15:0] sum [0:7];

    // Final product register
    reg [15:0] mul_out_reg;

    // Input enable pipeline
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            mul_en_out_reg <= 1'b0;
        end else begin
            mul_en_out_reg <= mul_en_in;
        end
    end

    // Input data pipeline: sample inputs when enabled
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            mul_a_reg <= 8'd0;
            mul_b_reg <= 8'd0;
        end else begin
            if (mul_en_in) begin
                mul_a_reg <= mul_a;
                mul_b_reg <= mul_b;
            end
        end
    end

    // Generate partial products
    genvar i;
    generate
        for (i = 0; i < 8; i = i + 1) begin : gen_pp
            assign temp[i] = mul_b_reg[i] ? {mul_a_reg, 8'b0} : 16'b0;
        end
    endgenerate

    // Pipeline stage for partial sums
    // Each sum register accumulates partial products in a tree-like fashion
    // For simplicity, we'll use a sequential addition pipeline over 8 stages
    // sum[i] holds the accumulated sum of partial products from 0 to i

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            for (int i = 0; i < 8; i = i + 1) begin
                sum[i] <= 16'd0;
            end
        end else begin
            // sum[0] = temp[0]
            sum[0] <= temp[0];
            // sum[i] = sum[i-1] + temp[i] for i = 1..7
            for (int i = 1; i < 8; i = i + 1) begin
                sum[i] <= sum[i-1] + temp[i];
            end
        end
    end

    // Final product calculation
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            mul_out_reg <= 16'd0;
        end else begin
            mul_out_reg <= sum[7];
        end
    end

    // Output enable
    assign mul_en_out = mul_en_out_reg;

    // Output product
    assign mul_out = mul_en_out ? mul_out_reg : 16'd0;

endmodule
