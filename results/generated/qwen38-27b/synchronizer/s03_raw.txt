```verilog
module synchronizer (
    input  wire        clk_a,
    input  wire        clk_b,
    input  wire        arstn,
    input  wire        brstn,
    input  wire [3:0]  data_in,
    input  wire        data_en,
    output reg  [3:0]  dataout
);

    // Data Register (clocked by clk_a)
    reg [3:0] data_reg;

    // Enable Data Register (clocked by clk_a)
    reg       en_data_reg;

    // Enable Control Registers (clocked by clk_b)
    reg       en_clap_one;
    reg       en_clap_two;

    // Data register update on clk_a positive edge
    always @(posedge clk_a or negedge arstn) begin
        if (!arstn) begin
            data_reg <= 4'd0;
        end else begin
            data_reg <= data_in;
        end
    end

    // Enable data register update on clk_a positive edge
    always @(posedge clk_a or negedge arstn) begin
        if (!arstn) begin
            en_data_reg <= 1'b0;
        end else begin
            en_data_reg <= data_en;
        end
    end

    // Enable control registers update on clk_b positive edge
    always @(posedge clk_b or negedge brstn) begin
        if (!brstn) begin
            en_clap_one <= 1'b0;
            en_clap_two <= 1'b0;
        end else begin
            en_clap_one <= en_data_reg;
            en_clap_two <= en_clap_one;
        end
    end

    // Output assignment on clk_b positive edge
    always @(posedge clk_b or negedge brstn) begin
        if (!brstn) begin
            dataout <= 4'd0;
        end else begin
            if (en_clap_two) begin
                dataout <= data_reg;
            end
            // else dataout retains its previous value (implicit by not assigning)
        end
    end

endmodule
```
