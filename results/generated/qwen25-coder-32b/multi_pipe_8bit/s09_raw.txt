```verilog
module multi_pipe_8bit (
    input wire clk,           // Clock signal
    input wire rst_n,         // Active-low reset signal
    input wire mul_en_in,     // Input enable signal
    input wire [7:0] mul_a,   // Multiplicand
    input wire [7:0] mul_b,   // Multiplier
    output reg mul_en_out,    // Output enable signal
    output reg [15:0] mul_out // Product output
);

    // Register to store the input enable signal
    reg mul_en_out_reg;

    // Registers to store the input values
    reg [7:0] mul_a_reg;
    reg [7:0] mul_b_reg;

    // Temporary wires to store partial products
    wire [7:0] temp [0:7];

    // Registers to store partial sums
    reg [15:0] sum [0:7];

    // Register to store the final product
    reg [15:0] mul_out_reg;

    // Input Control
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            mul_en_out_reg <= 0;
        end else begin
            mul_en_out_reg <= mul_en_in;
        end
    end

    // Input Registers
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            mul_a_reg <= 0;
            mul_b_reg <= 0;
        end else if (mul_en_in) begin
            mul_a_reg <= mul_a;
            mul_b_reg <= mul_b;
        end
    end

    // Partial Product Generation
    genvar i;
    generate
        for (i = 0; i < 8; i = i + 1) begin : partial_product_gen
            assign temp[i] = mul_b_reg[i] ? mul_a_reg : 8'b0;
        end
    endgenerate

    // Partial Sum Calculation
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            sum[0] <= temp[0];
        end else begin
            sum[0] <= temp[0];
        end
    end

    genvar j;
    generate
        for (j = 1; j < 8; j = j + 1) begin : partial_sum_gen
            always @(posedge clk or negedge rst_n) begin
                if (!rst_n) begin
                    sum[j] <= 0;
                end else begin
                    sum[j] <= sum[j-1] + ({8'b0, temp[j]});
                end
            end
        end
    endgenerate

    // Final Product Calculation
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            mul_out_reg <= 0;
        end else begin
            mul_out_reg <= sum[7];
        end
    end

    // Output Assignment
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            mul_en_out <= 0;
            mul_out <= 0;
        end else begin
            mul_en_out <= mul_en_out_reg;
            mul_out <= mul_en_out_reg ? mul_out_reg : 0;
        end
    end

endmodule
```

### Explanation:
1. **Input Control:**
   - `mul_en_out_reg` stores the input enable signal.
   - On the positive edge of the clock, `mul_en_out_reg` is updated with the value of `mul_en_in`.

2. **Input Registers:**
   - `mul_a_reg` and `mul_b_reg` store the multiplicand and multiplier respectively.
   - These registers are updated only when `mul_en_in` is high.

3. **Partial Product Generation:**
   - `temp[i]` is generated based on the individual bits of `mul_b_reg` and `mul_a_reg`.
   - If the bit of `mul_b_reg` is 1, `temp[i]` is assigned `mul_a_reg`, otherwise it is 0.

4. **Partial Sum Calculation:**
   - `sum[0]` is initialized with `temp[0]`.
   - For each subsequent bit position, `sum[i]` is calculated by adding the previous sum with the shifted partial product.

5. **Final Product Calculation:**
   - `mul_out_reg` stores the final product, which is the value of `sum[7]`.

6. **Output Assignment:**
   - `mul_en_out` is derived from `mul_en_out_reg`.
   - `mul_out` is assigned the value of `mul_out_reg` if `mul_en_out` is high, otherwise it is set to 0.

This design ensures that the multiplication process is pipelined, allowing for higher performance by overlapping the stages of computation.
