```verilog
module width_8to16 (

    always @(posedge clk or negedge rst_n) begin

            // Reset state
            valid_out <= 0;

            first_data_stored <= 0;

                data_lock <= data_in;
                first_data_stored <= 1;

                // Concatenate the first 8-bit data with the second 8-bit data
                data_out <= {data_lock, data_in};
                valid_out <= 1;
                first_data_stored <= 0;
            end
        end else begin
            // No new valid data, keep the output valid signal low
            valid_out <= 0;
        end
    end

endmodule
```

### Explanation:
1. **Registers**:

   - `first_data_stored`: This flag indicates whether the first 8-bit data has been stored.

2. **Synchronous Block**:
   - The `always @(posedge clk or negedge rst_n)` block ensures that the logic is executed on the positive edge of the clock or the negative edge of the reset signal.
   - On reset (`!rst_n`), all registers and flags are cleared.

   - If `valid_in` is low, `valid_out` remains 0.
