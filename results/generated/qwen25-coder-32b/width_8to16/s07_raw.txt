            // Reset condition
            data_lock <= 8'b0;
            flag <= 0;
            valid_out <= 0;

                    data_out <= {data_lock, data_in};

                    flag <= 0;
                end
            end else begin
                // If no valid data, keep valid_out low
                valid_out <= 0;
            end
        end
    end

endmodule

3. **Output**:
   - `valid_out` is high only when the 16-bit data has been formed by concatenating two 8-bit inputs.
   - `data_out` contains the concatenated 16-bit data.
