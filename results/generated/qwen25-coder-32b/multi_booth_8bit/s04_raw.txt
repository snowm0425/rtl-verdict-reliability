                multiplicand <= {multiplicand[14:0], 1'b0};

                // Increment the counter
                ctr <= ctr + 1;
            end else begin
                // Multiplication is complete
                rdy <= 1'b1;
            end
        end
    end

endmodule

- **Multiplicand Shifting**: The `multiplicand` register is left-shifted by 1 bit in each iteration.
- **Booth Encoding**: The Booth encoding is applied to determine whether to add, subtract, or do nothing to the product register `p` based on the current and next bit of the `multiplier`.
