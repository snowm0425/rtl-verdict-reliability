module serial2parallel (

    output reg dout_valid            // Validity signal for the output data.
);

    always @(posedge clk or negedge rst_n) begin

            dout_parallel <= 8'b0;  // Clear the parallel output on reset.
            dout_valid <= 0;        // Clear the valid output on reset.
            cnt <= 3'b0;            // Reset the counter on reset.

                dout_parallel <= {din_serial, dout_parallel[7:1]};

                // Check if 8 bits have been received.
                if (cnt == 3'd7) begin
                    dout_valid <= 1;  // Set the valid output to 1.
                    cnt <= 3'b0;      // Reset the counter.
                end else begin
                    dout_valid <= 0;  // Keep the valid output to 0.
                end
            end else begin
                dout_valid <= 0;  // Keep the valid output to 0 if din_valid is 0.
            end
        end
    end

endmodule

   - `dout_valid`: Indicates the validity of the parallel output data.

2. **Internal Registers:**

   - `dout_valid`: Indicates the validity of the parallel data.
   - `cnt`: A 3-bit counter to count the number of serial bits received.

3. **Synchronous Logic:**
   - The logic is sensitive to the rising edge of `clk` and the falling edge of `rst_n` (active low reset).
   - On reset, all outputs are cleared, and the counter is reset to 0.
   - When `din_valid` is high, the module shifts the current `dout_parallel` left by 1 bit and appends the new `din_serial` bit.
   - The counter `cnt` is incremented with each valid input bit.

   - If `din_valid` is low, `dout_valid` remains low.
