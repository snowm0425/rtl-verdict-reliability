            for (int j = 0; j < size; j = j + 1) begin
                reg1[j] <= 8'b00000000;
            end
        end else begin

                end else begin
                    reg2[j] <= reg2[j-1] + reg1[j];
                end

        end
    end

    assign mul_out = final_product;
