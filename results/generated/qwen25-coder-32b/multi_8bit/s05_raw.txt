module multi_8bit (

    reg [7:0] multiplicand;  // Register to hold the multiplicand A
    reg [7:0] multiplier;    // Register to hold the multiplier B
    reg [3:0] count;         // Counter to iterate through each bit of B

    initial begin
        acc = 16'b0;
        multiplicand = A;
        multiplier = B;
        count = 4'b0;
    end

    always @(posedge clk or posedge reset) begin
        if (reset) begin
            acc <= 16'b0;
            multiplicand <= A;
            multiplier <= B;
            count <= 4'b0;
        end else if (count < 8) begin
            // Add multiplicand to accumulator if the current bit of multiplier is 1
            if (multiplier[0] == 1'b1) begin
                acc <= acc + {8'b0, multiplicand}; // Shift multiplicand left by 8 bits and add to acc
            end

            multiplicand <= multiplicand << 1;
            // Shift multiplier right by 1 bit
            multiplier <= multiplier >> 1;
            // Increment the counter
            count <= count + 1;
        end else begin
            // Load the final product into the output port
            product <= acc;
        end
    end

endmodule

   - `count`: A counter to keep track of the number of bits processed in the multiplier.

2. **Initialization**:

3. **Sequential Logic**:

   - For each bit of the multiplier, if the bit is 1, the multiplicand is added to the accumulator (after shifting it left by 8 bits to align with the current bit position).

   - The counter is incremented after each iteration.
   - Once all bits of the multiplier are processed (when `count` reaches 8), the final product is loaded into the `product` output port.
