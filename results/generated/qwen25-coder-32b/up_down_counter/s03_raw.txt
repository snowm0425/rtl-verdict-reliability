```verilog
module up_down_counter (
    input wire clk,        // Clock signal
    input wire reset,      // Reset signal
    input wire up_down,    // Control signal to determine counting direction
    output reg [15:0] count // 16-bit output representing the current counter value
);

always @(posedge clk or posedge reset) begin
    if (reset) begin
        // Reset the counter to zero when reset signal is active
        count <= 16'b0000000000000000;
    end else begin
        // Increment or decrement the counter based on the up_down signal
        if (up_down) begin
            // Increment the counter
            count <= count + 1;
        end else begin
            // Decrement the counter
            count <= count - 1;
        end
    end
end

endmodule
```

### Explanation:
- **Module Declaration**: The module is named `up_down_counter` and includes three input ports (`clk`, `reset`, `up_down`) and one output port (`count`).
- **Always Block**: The `always` block is triggered on the rising edge of the `clk` or `reset` signals.
  - **Reset Condition**: If the `reset` signal is active (`reset` is high), the counter is reset to zero.
  - **Counting Logic**: Depending on the value of `up_down`, the counter either increments or decrements:
    - If `up_down` is high (`1`), the counter increments by 1.
    - If `up_down` is low (`0`), the counter decrements by 1.
- **Output**: The `count` output reflects the current value of the counter.
