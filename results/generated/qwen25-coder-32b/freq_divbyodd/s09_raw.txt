```verilog
module freq_divbyodd #(
    parameter NUM_DIV = 5 // Must be an odd number
)(
    input clk,          // Input clock signal
    input rst_n,        // Active low reset signal
    output reg clk_div    // Divided clock output
);

    // Ensure NUM_DIV is an odd number
    initial begin
        if (NUM_DIV % 2 == 0) begin
            $fatal("NUM_DIV must be an odd number.");
        end
    end

    reg clk_div1, clk_div2;                  // Clock dividers for rising and falling edges

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            cnt1 <= 0;
            clk_div1 <= 0;
        end else begin
            if (cnt1 == (NUM_DIV - 1)) begin
                cnt1 <= 0;
                clk_div1 <= ~clk_div1;
            end else begin
                cnt1 <= cnt1 + 1;
            end
        end
    end

    // Counter for falling edge
    always @(negedge clk or negedge rst_n) begin
        if (!rst_n) begin
            cnt2 <= 0;
            clk_div2 <= 0;
        end else begin
            if (cnt2 == (NUM_DIV - 1)) begin
                cnt2 <= 0;
                clk_div2 <= ~clk_div2;
            end else begin
                cnt2 <= cnt2 + 1;
            end
        end
    end

    // Final divided clock output
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            clk_div <= 0;
        end else begin
            clk_div <= clk_div1 | clk_div2;
        end
    end

endmodule
```

2. **Initial Check**:

4. **Clock Dividers**:
   - `clk_div1` and `clk_div2` are used to toggle on the rising and falling edges of the clock, respectively.
   - When the counters reach `NUM_DIV - 1`, they reset to 0 and toggle the respective clock dividers.

5. **Final Output**:
   - The final divided clock output `clk_div` is derived by logically OR-ing `clk_div1` and `clk_div2` to account for both clock edges.

6. **Reset**:
   - The active low reset signal `rst_n` initializes the counters and clock divider outputs to 0.
