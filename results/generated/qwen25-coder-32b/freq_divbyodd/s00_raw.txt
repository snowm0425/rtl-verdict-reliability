                clk_div1 <= ~clk_div1;
            end

            // Toggle clk_div2 on falling edge
            if (cnt2 == (NUM_DIV / 2)) begin
                clk_div2 <= ~clk_div2;
            end
        end
    end

    // Generate final divided clock output

        clk_div = clk_div1 | clk_div2;
    end

endmodule
