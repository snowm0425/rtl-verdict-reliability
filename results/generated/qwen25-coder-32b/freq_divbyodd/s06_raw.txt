```verilog
module freq_divbyodd #(
    parameter NUM_DIV = 5  // Default divisor is 5, must be an odd number
) (
    input wire clk,       // Input clock signal
    input wire rst_n,     // Active low reset signal
    output wire clk_div   // Divided clock output
);

    // Ensure NUM_DIV is odd
    generate
        if (NUM_DIV % 2 == 0) begin : odd_check
            initial begin
                $error("NUM_DIV must be an odd number.");
            end
        end
    endgenerate

    // Counters to track the rising and falling edges of the clock
    reg [($clog2(NUM_DIV)-1):0] cnt1, cnt2;
    reg clk_div1, clk_div2;

    // Synchronous reset and counter logic
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            cnt1 <= 0;
            cnt2 <= 0;
            clk_div1 <= 0;
            clk_div2 <= 0;
        end else begin
            // Counter for rising edges
            if (cnt1 == NUM_DIV - 1) begin
                cnt1 <= 0;
                clk_div1 <= ~clk_div1;
            end else begin
                cnt1 <= cnt1 + 1;
            end

            // Counter for falling edges
            if (cnt2 == NUM_DIV - 1) begin
                cnt2 <= 0;
                clk_div2 <= ~clk_div2;
            end else begin
                cnt2 <= cnt2 + 1;
            end
        end
    end

    // Final divided clock output is the OR of both dividers
    assign clk_div = clk_div1 | clk_div2;

endmodule
```

### Explanation:
1. **Parameters and Ports**:
   - `NUM_DIV`: Parameter to specify the odd divisor. It defaults to 5.
   - `clk`: Input clock signal.
   - `rst_n`: Active low reset signal.
   - `clk_div`: Output divided clock signal.

2. **Odd Check**:
   - A generate block is used to check if `NUM_DIV` is odd. If it is even, an error is thrown.

3. **Counters**:
   - Two counters, `cnt1` and `cnt2`, are used to track the rising and falling edges of the clock.
   - The counters count up to `NUM_DIV - 1`.

4. **Clock Dividers**:
   - `clk_div1` and `clk_div2` are toggled when the respective counters reach `NUM_DIV - 1`.
   - The final divided clock output `clk_div` is the logical OR of `clk_div1` and `clk_div2`.

5. **Reset Logic**:
   - On active low reset (`rst_n` = 0), all counters and dividers are reset to their initial states.

This code ensures that the frequency is divided by an odd number specified by `NUM_DIV`, and it handles the reset condition properly.
