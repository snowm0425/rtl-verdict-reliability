```verilog
module synchronizer (

    input wire clk_b,          // Clock signal B
    input wire arstn,          // Active-low reset signal for clk_a
    input wire brstn,          // Active-low reset signal for clk_b
    input wire [3:0] data_in,  // 4-bit input data
    input wire data_en,        // Enable signal

    output reg [3:0] dataout   // 4-bit output data
);

    // Registers to store input data and enable signal
    reg [3:0] data_reg;
    reg en_data_reg;

    // Control registers for synchronization
    reg en_clap_one;
    reg en_clap_two;

    // Data register update logic with reset
    always @(posedge clk_a or negedge arstn) begin
        if (!arstn) begin
            data_reg <= 4'b0000;
        end else begin
            data_reg <= data_in;
        end
    end

    // Enable data register update logic with reset
    always @(posedge clk_a or negedge arstn) begin
        if (!arstn) begin
            en_data_reg <= 1'b0;
        end else begin
            en_data_reg <= data_en;
        end
    end

    // Control registers update logic with reset
    always @(posedge clk_b or negedge brstn) begin
        if (!brstn) begin
            en_clap_one <= 1'b0;
            en_clap_two <= 1'b0;
        end else begin
            en_clap_one <= en_data_reg;
            en_clap_two <= en_clap_one;
        end
    end

    // Output assignment logic with reset
    always @(posedge clk_b or negedge brstn) begin
        if (!brstn) begin
            dataout <= 4'b0000;
        end else if (en_clap_two) begin
            dataout <= data_reg;
        end
    end

endmodule
```

### Explanation:

1. **Data Register (`data_reg`)**:
   - On the rising edge of `clk_a`, `data_reg` is updated with the value of `data_in`.
   - On the falling edge of `arstn`, `data_reg` is reset to `4'b0000`.

2. **Enable Data Register (`en_data_reg`)**:
   - On the rising edge of `clk_a`, `en_data_reg` is updated with the value of `data_en`.
   - On the falling edge of `arstn`, `en_data_reg` is reset to `1'b0`.

3. **Enable Control Registers (`en_clap_one` and `en_clap_two`)**:
   - On the rising edge of `clk_b`, `en_clap_one` is updated with the value of `en_data_reg`, and `en_clap_two` is updated with the previous value of `en_clap_one`.
   - On the falling edge of `brstn`, both `en_clap_one` and `en_clap_two` are reset to `1'b0`.

4. **Output Assignment (`dataout`)**:
   - On the rising edge of `clk_b`, if `brstn` is high and `en_clap_two` is active, `dataout` is assigned the value of `data_reg`.
   - On the falling edge of `brstn`, `dataout` is reset to `4'b0000`.

This module ensures that the input data `data_in` is properly synchronized to the clock domain of `clk_b` with a two-cycle delay, and only when `data_en` is high.
