```verilog
module synchronizer (

    input wire clk_b,          // Clock signal B
    input wire arstn,          // Active-low reset signal A
    input wire brstn,          // Active-low reset signal B
    input wire [3:0] data_in,  // Input data signal of size 4 bits
    input wire data_en,        // Input enable signal

    output reg [3:0] dataout   // Output data signal of size 4 bits
);

    // Register to store input data
    reg [3:0] data_reg;

    // Register to store enable signal
    reg en_data_reg;

    // Enable control registers
    reg en_clap_one;
    reg en_clap_two;

    // Data register update on positive edge of clk_a or falling edge of arstn
    always @(posedge clk_a or negedge arstn) begin
        if (!arstn) begin
            data_reg <= 4'b0;
        end else begin
            data_reg <= data_in;
        end
    end

    // Enable register update on positive edge of clk_a or falling edge of arstn
    always @(posedge clk_a or negedge arstn) begin
        if (!arstn) begin
            en_data_reg <= 1'b0;
        end else begin
            en_data_reg <= data_en;
        end
    end

    // Enable control registers update on positive edge of clk_b or falling edge of brstn
    always @(posedge clk_b or negedge brstn) begin
        if (!brstn) begin
            en_clap_one <= 1'b0;
            en_clap_two <= 1'b0;
        end else begin
            en_clap_one <= en_data_reg;
            en_clap_two <= en_clap_one;
        end
    end

    // Output assignment on positive edge of clk_b or falling edge of brstn
    always @(posedge clk_b or negedge brstn) begin
        if (!brstn) begin
            dataout <= 4'b0;
        end else if (en_clap_two) begin
            dataout <= data_reg;
        end
    end

endmodule
```

   - Updated on the positive edge of `clk_a` or the falling edge of `arstn`.
   - If `arstn` is low, it sets `data_reg` to `4'b0`.
   - If `arstn` is high, it updates `data_reg` with the value of `data_in`.

2. **Enable Data Register (`en_data_reg`)**:
   - Updated on the positive edge of `clk_a` or the falling edge of `arstn`.
   - If `arstn` is low, it sets `en_data_reg` to `1'b0`.
   - If `arstn` is high, it updates `en_data_reg` with the value of `data_en`.

3. **Enable Control Registers (`en_clap_one` and `en_clap_two`)**:
   - Updated on the positive edge of `clk_b` or the falling edge of `brstn`.
   - If `brstn` is low, it sets both `en_clap_one` and `en_clap_two` to `1'b0`.
   - If `brstn` is high, it updates `en_clap_one` with the value of `en_data_reg` and `en_clap_two` with the previous value of `en_clap_one`.

4. **Output Assignment (`dataout`)**:
   - Updated on the positive edge of `clk_b` or the falling edge of `brstn`.
   - If `brstn` is low, it sets `dataout` to `4'b0`.
   - If `brstn` is high and `en_clap_two` is active, it assigns `data_reg` to `dataout`.
   - If `en_clap_two` is inactive, `dataout` retains its previous value.

This module ensures that `data_in` is synchronized to `clk_b` and only updated when `data_en` is high for at least 3 `clk_b` cycles, as per your requirements.
