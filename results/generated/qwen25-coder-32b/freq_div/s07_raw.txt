```verilog
module freq_div (

    input RST,         // Active high reset signal

    output reg CLK_1   // 1 MHz output clock

    // Counters for CLK_10 and CLK_1 generation
    reg [2:0] cnt_10;  // Counter for 10 MHz clock (range 0 to 4)
    reg [5:0] cnt_100; // Counter for 1 MHz clock (range 0 to 49)

        if (RST) begin
            CLK_50 <= 0;
        end else begin
            CLK_50 <= ~CLK_50; // Toggle CLK_50 on each rising edge of CLK_in

            CLK_10 <= 0;
            cnt_10 <= 0;
        end else begin
            if (cnt_10 == 4) begin // Toggle CLK_10 every 5 cycles (100 MHz / 5 = 20 MHz, then toggle every other cycle to get 10 MHz)
                CLK_10 <= ~CLK_10;
                cnt_10 <= 0;
            end else begin
                cnt_10 <= cnt_10 + 1;
            end
        end
    end

    // CLK_1 generation
    always @(posedge CLK_in or posedge RST) begin
        if (RST) begin
            CLK_1 <= 0;
            cnt_100 <= 0;
        end else begin
            if (cnt_100 == 49) begin // Toggle CLK_1 every 50 cycles (100 MHz / 50 = 2 MHz, then toggle every other cycle to get 1 MHz)

                cnt_100 <= 0;

                cnt_100 <= cnt_100 + 1;
            end
        end
    end

endmodule
```

### Explanation:
- **CLK_50 Generation**: The `CLK_50` signal is toggled on every rising edge of the `CLK_in` signal. Since the input clock is 100 MHz, toggling every cycle results in a 50 MHz output.
  
- **CLK_10 Generation**: The `cnt_10` counter counts up to 4 (5 cycles in total). When it reaches 4, it toggles the `CLK_10` signal and resets the counter. This effectively divides the 100 MHz clock by 10 to produce a 10 MHz output.

- **CLK_1 Generation**: The `cnt_100` counter counts up to 49 (50 cycles in total). When it reaches 49, it toggles the `CLK_1` signal and resets the counter. This effectively divides the 100 MHz clock by 100 to produce a 1 MHz output.

This module uses edge-triggered `always` blocks to ensure that the clock signals are generated correctly on the rising edge of the input clock or when the reset signal is active.
