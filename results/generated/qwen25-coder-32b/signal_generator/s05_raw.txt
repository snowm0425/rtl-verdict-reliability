```verilog
module signal_generator (

    reg state; // 1-bit state register (0 for increment, 1 for decrement)

    always @(posedge clk or negedge rst_n) begin

            wave <= 0;

            case (state)

                        state <= 0; // Transition to increment state

                        wave <= wave - 1; // Decrement wave
                    end
                end
                default: begin
                    // Default case: should never reach here
                    state <= 0;
                    wave <= 0;
                end
            endcase
        end
    end

endmodule
```

### Explanation:

  - `rst_n`: The active-low reset signal. When `rst_n` is 0, the module is reset.

- **Outputs:**
  - `wave`: A 5-bit register that represents the generated waveform signal.

- **Registers:**
  - `state`: A 1-bit register that controls whether the waveform is being incremented or decremented.
  - `wave`: A 5-bit register that holds the current value of the waveform.
