module signal_generator (

    input rst_n,      // Active-low reset signal
    output reg [4:0] wave // 5-bit output waveform signal
);

    typedef enum logic { 
        INCREMENT = 0, 
        DECREMENT = 1 
    } state_t;

    // State register
    reg state;

    // Initial block to set initial state and wave value
    initial begin
        state = INCREMENT;
        wave = 5'b00000;
    end

    // Always block to handle state transitions and wave generation
    always @(posedge clk or negedge rst_n) begin
        if (~rst_n) begin
            // Reset condition: clear state and wave
            state <= INCREMENT;
            wave <= 5'b00000;
        end else begin
            // State machine logic
            case (state)

                    if (wave == 5'b11111) begin
                        // If wave reaches 31, transition to DECREMENT
                        state <= DECREMENT;
                        wave <= wave - 1;
                    end else begin
                        // Otherwise, increment wave
                        wave <= wave + 1;
                    end
                end
                DECREMENT: begin
                    if (wave == 5'b00000) begin
                        // If wave reaches 0, transition to INCREMENT
                        state <= INCREMENT;
                        wave <= wave + 1;
                    end else begin
                        // Otherwise, decrement wave
                        wave <= wave - 1;
                    end
                end
                default: begin
                    // Default case: should not reach here
                    state <= INCREMENT;
                    wave <= 5'b00000;
                end
            endcase
        end
    end

endmodule

### Explanation:
1. **State Definition**: The `state_t` enum defines two states: `INCREMENT` and `DECREMENT`.
2. **State and Wave Registers**:

   - `wave` register is a 5-bit register that holds the current value of the waveform.
3. **Initial Block**: Initializes the `state` to `INCREMENT` and `wave` to 0 when the simulation starts.
4. **Always Block**:

   - On reset (`rst_n` is low), the `state` and `wave` are reset to their initial values.
   - On clock edge, the state machine logic is executed:

     - If in `DECREMENT` state and `wave` reaches 0, transition to `INCREMENT` and increment `wave`.
     - Otherwise, continue incrementing or decrementing `wave` based on the current state.
