module adder_pipe_64bit (

    input wire rst_n,
    input wire i_en,
    input wire [63:0] adda,
    input wire [63:0] addb,
    output reg [64:0] result,
    output reg o_en
);

    localparam WIDTH = 64 / STAGES;

    reg [WIDTH-1:0] adda_reg [0:STAGES];

                    if (!rst_n) begin
                        adda_reg[0] <= 0;
                        addb_reg[0] <= 0;
                        sum_reg[0] <= 0;
                        carry_reg[0] <= 0;

                        addb_reg[0] <= addb[WIDTH-1:0];
                        {sum_reg[0], carry_reg[0]} <= adda[WIDTH-1:0] + addb[WIDTH-1:0];
                    end
                end
            end
            // Middle pipeline stages
            else if (i < STAGES) begin
                always @(posedge clk or negedge rst_n) begin
                    if (!rst_n) begin
                        adda_reg[i] <= 0;
                        addb_reg[i] <= 0;
                        sum_reg[i] <= 0;

                end
            end

                    if (!rst_n) begin
                        sum_reg[i] <= 0;
                        o_en <= 0;

                        sum_reg[i] <= adda[63:(STAGES-1)*WIDTH] + addb[63:(STAGES-1)*WIDTH] + carry_reg[i-1];

    always @(posedge clk or negedge rst_n) begin

### Explanation:
1. **Parameters**: The module is parameterized with `STAGES` and `WIDTH` to define the number of pipeline stages and the width of each stage, respectively.
2. **Registers**: Internal registers (`adda_reg`, `addb_reg`, `sum_reg`, `carry_reg`) are used to hold the operands and intermediate results at each stage.
3. **Pipeline Stages**: The `generate` block creates the pipeline stages. Each stage processes a segment of the 64-bit input operands and calculates the partial sum and carry.
