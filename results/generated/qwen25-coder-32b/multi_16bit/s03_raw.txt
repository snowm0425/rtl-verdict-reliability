            i <= 0;
            areg <= 16'b0;
            breg <= 16'b0;
            yout_r <= 32'b0;
            done_r <= 0;

                    i <= i + 1;
                end else begin
                    i <= 0;
                end

                done_r <= 1;
            end else if (i == 17) begin

                areg <= ain;
                breg <= bin;
