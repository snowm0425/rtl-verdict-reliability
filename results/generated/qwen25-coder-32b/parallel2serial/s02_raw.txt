            // Reset the counter, data register, and valid signal
            cnt <= 2'b00;
            data <= 4'b0000;

            if (cnt == 2'b11) begin

                valid_out <= 1;
            end else begin

                valid_out <= 0;
                // Increment the counter
                cnt <= cnt + 1;
                // Shift the data register left by 1 bit
                data <= {data[2:0], data[3]};
            end
            // Assign the most significant bit of the data register to the serial output
            dout <= data[3];
        end
    end

endmodule

  - `cnt`: A 2-bit counter that tracks the position of the bit currently being output.
  
- **Reset Logic**:

- **Normal Operation**:
