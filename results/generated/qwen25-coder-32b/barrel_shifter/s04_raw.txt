```verilog
module mux2X1 (
    input sel,       // 1-bit select signal
    input a,         // 1-bit first input
    input b,         // 1-bit second input
    output reg y     // 1-bit output
);

always @(*) begin
    if (sel)
        y = b;
    else
        y = a;
end

endmodule
```
