```verilog
module calendar (
    input wire CLK,    // Clock input
    input wire RST,    // Active high reset signal
    output reg [5:0] Hours, // 6-bit output representing the current hours
    output reg [5:0] Mins,  // 6-bit output representing the current minutes
    output reg [5:0] Secs   // 6-bit output representing the current seconds
);

// Always block to update seconds
always @(posedge CLK or posedge RST) begin
    if (RST) begin
        Secs <= 6'b000000;  // Reset seconds to 0
    end else if (Secs == 6'b111011) begin  // Check if seconds == 59
        Secs <= 6'b000000;  // Wrap around to 0
    end else begin
        Secs <= Secs + 1;   // Increment seconds
    end
end

// Always block to update minutes
always @(posedge CLK or posedge RST) begin
    if (RST) begin
        Mins <= 6'b000000;  // Reset minutes to 0
    end else if (Secs == 6'b111011 && Mins == 6'b111011) begin  // Check if seconds == 59 and minutes == 59
        Mins <= 6'b000000;  // Wrap around to 0
    end else if (Secs == 6'b111011) begin  // Check if seconds == 59
        Mins <= Mins + 1;   // Increment minutes
    end
end

// Always block to update hours
always @(posedge CLK or posedge RST) begin
    if (RST) begin
        Hours <= 6'b000000; // Reset hours to 0
    end else if (Hours == 6'b010111 && Mins == 6'b111011 && Secs == 6'b111011) begin  // Check if hours == 23 and minutes == 59 and seconds == 59
        Hours <= 6'b000000; // Wrap around to 0
    end else if (Mins == 6'b111011 && Secs == 6'b111011) begin  // Check if minutes == 59 and seconds == 59
        Hours <= Hours + 1; // Increment hours
    end
end

endmodule
```

### Explanation:
1. **Seconds Update**:
   - The `Secs` register is incremented on each clock cycle.
   - When `Secs` reaches 59 (binary `111011`), it wraps around to 0.
   - On a reset, `Secs` is set to 0.

2. **Minutes Update**:
   - The `Mins` register is incremented when `Secs` reaches 59.
   - When both `Secs` and `Mins` reach 59, `Mins` wraps around to 0.
   - On a reset, `Mins` is set to 0.

3. **Hours Update**:
   - The `Hours` register is incremented when both `Mins` and `Secs` reach 59.
   - When `Hours`, `Mins`, and `Secs` reach 23, 59, and 59 respectively, `Hours` wraps around to 0.
   - On a reset, `Hours` is set to 0.

This code implements a simple perpetual calendar with hours ranging from 0 to 23, minutes and seconds ranging from 0 to 59.
