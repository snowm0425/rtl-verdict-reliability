```verilog
module square_wave (
    input clk,          // Clock signal that drives the module
    input [7:0] freq,   // Frequency control, specifying how often the square wave toggles
    output reg wave_out // Square wave output signal
);

    reg [7:0] count;    // Counter register used to track cycles between wave toggles

    always @(posedge clk) begin
        if (count == (freq - 1)) begin
            count <= 0;           // Reset count to 0
            wave_out <= ~wave_out; // Toggle wave_out signal
        end else begin
            count <= count + 1;   // Increment count
        end
    end

endmodule
```

3. **Internal Register:**

4. **Counter Operation:**
   - The `always @(posedge clk)` block ensures that the logic inside it executes on every rising edge of the clock signal.
   - If `count` equals `freq - 1`, it resets `count` to 0 and toggles `wave_out` using the bitwise NOT operator (`~`).
   - If `count` is not equal to `freq - 1`, it simply increments `count` by 1.

5. **Square Wave Generation:**
   - The module generates a square wave by flipping the `wave_out` signal at the rate determined by the `freq` input. Higher `freq` values result in a slower toggle rate, effectively lowering the output frequency.
