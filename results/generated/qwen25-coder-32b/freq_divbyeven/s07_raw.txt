module freq_diveven #(

) (

    output reg clk_div     // Divided clock output signal
);

    // Ensure NUM_DIV is an even number
    generate

            initial begin
                $error("NUM_DIV must be an even number.");

            end
        end

    // Counter to count the clock cycles
    reg [3:0] cnt;  // Assuming NUM_DIV will not exceed 16 (2^4)

    // State machine to handle the frequency division
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            cnt <= 4'b0000;
            clk_div <= 1'b0;
        end else begin
            if (cnt < (NUM_DIV / 2 - 1)) begin
                cnt <= cnt + 1;
            end else begin
                cnt <= 4'b0000;
                clk_div <= ~clk_div;
            end
        end
    end

endmodule

4. **Always Block**: The always block handles the logic for frequency division:

   - On a positive edge of `clk`, the counter increments until it reaches `(NUM_DIV / 2 - 1)`. At this point, the counter resets to zero and the `clk_div` signal is toggled, effectively dividing the input clock frequency by `NUM_DIV`.

This code ensures that the frequency of the output clock (`clk_div`) is divided by the specified even number (`NUM_DIV`).
