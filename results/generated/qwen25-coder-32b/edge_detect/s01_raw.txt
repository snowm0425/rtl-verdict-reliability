module edge_detect (

    output reg down       // Output signal indicating a falling edge

    // Synchronize the input signal 'a' with the clock
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            a_reg <= 0;
            rise <= 0;
            down <= 0;
        end else begin
            a_reg <= a;
            // Detect rising edge
            rise <= (a && !a_reg);
            // Detect falling edge
            down <= (!a && a_reg);
        end
    end

endmodule
