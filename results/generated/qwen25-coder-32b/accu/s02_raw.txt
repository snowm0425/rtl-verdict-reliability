```verilog
module accu (

    output reg [9:0] data_out // 10-bit output data representing the accumulated sum
);

    reg [1:0] count_reg;     // Counter for the number of valid inputs received

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            // Reset the accumulator and counter
            acc_reg <= 10'b0;
            count_reg <= 2'b0;
            valid_out <= 1'b0;
        end else begin

                // Accumulate the incoming data
                acc_reg <= acc_reg + data_in;
                count_reg <= count_reg + 1'b1;

                    // Output the accumulated data
                    data_out <= acc_reg;
                    valid_out <= 1'b1;

                    // Reset the accumulator and counter for the next set of data
                    acc_reg <= 10'b0;
                    count_reg <= 2'b0;
                end else begin
                    valid_out <= 1'b0;
                end
            end else begin
                valid_out <= 1'b0;
            end
        end
    end

endmodule
```

### Explanation:
1. **Reset Handling**: The module resets the accumulator (`acc_reg`) and counter (`count_reg`) when `rst_n` is low.
2. **Accumulation**: When `valid_in` is high, the `data_in` is added to the accumulator (`acc_reg`).
3. **Counting**: The `count_reg` keeps track of the number of valid inputs received. It is incremented each time valid data is received.
4. **Output**: Once 4 valid inputs have been received (i.e., `count_reg` reaches 3, which is `2'b11` in binary), the accumulated sum is output on `data_out`, and `valid_out` is set to high for one clock cycle.
5. **Reset After Output**: After outputting the accumulated sum, both the accumulator and the counter are reset to start the process again.

This module ensures that the accumulation and output process is synchronized with the clock and properly resets when the reset signal is active.
