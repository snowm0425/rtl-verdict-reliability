module traffic_light (

            state <= idle;
        end else begin

                    state <= s1_red;
                end
                s1_red: begin

                        state <= s3_green;
                end
                s2_yellow: begin
                    if (cnt == 0)
                        state <= s1_red;
                end

                        state <= s2_yellow;
                end

                    state <= idle;
                end

        end
    end

        end
    end
