module traffic_light (

    output reg yellow,        // Output signal representing the yellow light
    output reg green          // Output signal representing the green light
);

    // State definitions
    typedef enum reg [1:0] {
        idle = 2'b00,
        s1_red = 2'b01,
        s2_yellow = 2'b10,
        s3_green = 2'b11
    } state_t;

    // Registers
    reg [7:0] cnt;            // 8-bit register for timing
    state_t state, next_state; // State registers
    reg p_red, p_yellow, p_green; // Previous values of the output signals

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            state <= idle;
        end else begin
            state <= next_state;
        end
    end

    always @(*) begin
        case (state)
            idle: begin
                if (cnt == 0)
                    next_state = s1_red;
                else
                    next_state = idle;
            end
            s1_red: begin
                if (cnt == 0)
                    next_state = s3_green;
                else
                    next_state = s1_red;
            end
            s2_yellow: begin
                if (cnt == 0)
                    next_state = s1_red;
                else
                    next_state = s2_yellow;
            end
            s3_green: begin
                if (cnt == 0 || (pass_request && cnt > 10))
                    next_state = s2_yellow;
                else
                    next_state = s3_green;
            end
            default: next_state = idle;
        endcase
    end

        if (!rst_n) begin
            cnt <= 10;

            case (state)
                idle: begin
                    cnt <= 10;
                end
                s1_red: begin
                    if (cnt == 0)
                        cnt <= 60;
                    else
                        cnt <= cnt - 1;
                end
                s2_yellow: begin
                    if (cnt == 0)
                        cnt <= 5;
                    else
                        cnt <= cnt - 1;
                end

                    if (pass_request && cnt > 10)
                        cnt <= 10;
                    else if (cnt == 0)
                        cnt <= 5;
                    else
                        cnt <= cnt - 1;
                end
            endcase
        end
    end

    // Output signal logic
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            red <= 0;
            yellow <= 0;
            green <= 0;
        end else begin
            red <= p_red;
            yellow <= p_yellow;
            green <= p_green;
        end
    end

    always @(*) begin
        case (state)
            idle: begin
                p_red = 0;
                p_yellow = 0;
                p_green = 0;

                p_red = 0;
                p_yellow = 1;
                p_green = 0;

                p_yellow = 0;
                p_green = 1;
            end

            end
        endcase
    end

    // Assign the internal counter to the output
    assign clock = cnt;

endmodule
